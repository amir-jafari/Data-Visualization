# FastAPI course -- everything in fastapi/.
#
# On the course server, FastAPI is the ONLY package you need to add. Install it
# without letting pip touch anything else:
#
#     pip3 install --no-deps fastapi==0.141.1
#
# --no-deps is safe here because every dependency is already installed and
# already new enough:
#
#     starlette          >=0.46.0   you have 1.3.1
#     pydantic           >=2.9.0    you have 2.12.5
#     typing-extensions  >=4.8.0    you have 4.15.0
#     typing-inspection  >=0.4.2    you have 0.4.2
#
# Do NOT pin an older FastAPI. Versions before ~0.119 cap starlette at <0.42,
# so pip would DOWNGRADE starlette 1.3.1 and take other things with it.

fastapi>=0.119            # the only genuinely new package

# --- already on the course server; listed so a fresh machine gets them ------
# Unpinned on purpose: pip leaves an installed package alone unless you pass
# -U, so `pip install -r` here will not change versions you already have.
uvicorn                   # the web server that actually listens on the port
pydantic                  # validation and models
pydantic-settings         # settings from the environment (structure/settings.py)
python-multipart          # file uploads and form fields (files/upload.py)
httpx                     # used by TestClient (testing/*)

# --- also already present; used by the capstone in project/ ----------------
pandas
scikit-learn              # the model, and the built-in fallback datasets
boto3                     # optional: serves the course S3 datasets when keys work
requests                  # the Streamlit client calls the API with it
streamlit                 # the client page itself
